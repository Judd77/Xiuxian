using System;
using System.IO;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.EventSystems;
using UnityEngine.InputSystem.UI;
using WenXinJie.Core;
namespace WenXinJie.UI
{
 [Serializable] public sealed class IntroData { public string[] lines; }
 [Serializable] public sealed class PrototypeSave { public int version; public string session; public int line; public string message; }
 public sealed class FirstDayPrototype : MonoBehaviour
 {
  public TextAsset introAsset;
  private FirstDaySession session;
  private IntroData intro;
  private int line;
  private bool menu=true;
  private string message=string.Empty;
  private Font font;
  private Text title,status,body,notice;
  private Button next;
  private Button[] actions=new Button[4];
  public int RemainingActions => session.Snapshot().actionsLeft;
  #if UNITY_EDITOR
  [NonSerialized] public string testSaveDirectory;
#endif
  public string SavePath
  {
   get
   {
#if UNITY_EDITOR
    if(!string.IsNullOrEmpty(testSaveDirectory))return Path.Combine(testSaveDirectory,"quick.json");
#endif
    return Path.Combine(Application.persistentDataPath,"WenXinJiePrototype","quick.json");
   }
  }
  private void Awake()
  {
   intro=JsonUtility.FromJson<IntroData>(introAsset.text);
   if(intro==null||intro.lines==null||intro.lines.Length==0)throw new Exception("Invalid intro");
   session=new FirstDaySession();
   font=Font.CreateDynamicFontFromOSFont(new[]{"Microsoft YaHei","SimHei","Arial"},28);
   Build();Render();
  }
  private RectTransform Box(Transform parent,string name,float x,float y,float w,float h)
  {
   var go=new GameObject(name,typeof(RectTransform));go.transform.SetParent(parent,false);
   var r=(RectTransform)go.transform;r.anchorMin=r.anchorMax=new Vector2(0,1);r.pivot=new Vector2(0,1);
   r.anchoredPosition=new Vector2(x,-y);r.sizeDelta=new Vector2(w,h);return r;
  }
  private Text Label(Transform parent,string text,float x,float y,float w,float h,int size)
  {
   var t=Box(parent,"Text",x,y,w,h).gameObject.AddComponent<Text>();t.font=font;t.fontSize=size;
   t.text=text;t.color=new Color(.91f,.88f,.79f);t.supportRichText=false;t.raycastTarget=false;return t;
  }
  private Button Button(Transform parent,string text,float x,float y,float w,Action callback)
  {
   var r=Box(parent,text,x,y,w,84);var image=r.gameObject.AddComponent<Image>();image.color=new Color(.15f,.27f,.28f);
   var b=r.gameObject.AddComponent<Button>();b.targetGraphic=image;
   var label=Label(r,text,18,10,w-36,64,29);label.alignment=TextAnchor.MiddleLeft;
   b.onClick.AddListener(()=>callback());return b;
  }
  private void Build()
  {
   var c=new GameObject("PrototypeCanvas",typeof(RectTransform),typeof(Canvas),typeof(CanvasScaler),typeof(GraphicRaycaster));
   c.transform.SetParent(transform,false);c.GetComponent<Canvas>().renderMode=RenderMode.ScreenSpaceOverlay;
   var scaler=c.GetComponent<CanvasScaler>();scaler.uiScaleMode=CanvasScaler.ScaleMode.ScaleWithScreenSize;
   scaler.referenceResolution=new Vector2(1920,1080);scaler.screenMatchMode=CanvasScaler.ScreenMatchMode.Expand;
   var frame=Box(c.transform,"DesignFrame",0,0,1920,1080);
   frame.anchorMin=frame.anchorMax=new Vector2(.5f,.5f);frame.pivot=new Vector2(.5f,.5f);
   var background=Box(frame,"Background",0,0,1920,1080).gameObject.AddComponent<Image>();
   background.color=new Color(.04f,.08f,.10f);background.raycastTarget=false;
   Label(frame,"问心劫 / 山门夜雨",84,42,1260,84,55);
   Label(frame,"第一日规则原型 · 临时界面",1344,66,504,54,24);
   status=Label(frame,string.Empty,84,150,1752,72,29);
   title=Label(frame,string.Empty,84,264,1752,72,41);
   body=Label(frame,string.Empty,84,372,1752,264,36);
   next=Button(frame,"继续",1380,666,456,Advance);
   string[] names={"练剑 / 修为+2","查名册 / 花费1灵石，获得证据","陪林月 / 信任+2","药庐做工 / 灵石+2"};
   for(int i=0;i<4;i++){int n=i;actions[i]=Button(frame,names[i],84+(i%2)*894,666+(i/2)*108,858,()=>Choose((DayAction)n));}
   Button(frame,"快速存档",84,918,288,Save);
   Button(frame,"快速读档",396,918,288,Load);
   Button(frame,"返回标题",708,918,288,()=>{menu=true;Render();});
   notice=Label(frame,"单槽测试存档；新游戏不会自动覆盖旧存档。",84,1020,1752,48,24);
   if(EventSystem.current==null){var e=new GameObject("EventSystem",typeof(EventSystem),typeof(InputSystemUIInputModule));e.transform.SetParent(transform,false);e.GetComponent<InputSystemUIInputModule>().AssignDefaultActions();}
  }
  public void Begin(){session=new FirstDaySession();line=0;message=string.Empty;menu=false;Render();}
  public void Advance()
  {
   if(menu){Begin();return;}
   if(line<intro.lines.Length){line++;Render();return;}
   var s=session.Snapshot();
   if(s.resolved){Begin();return;}
   if(s.actionsLeft==0){session.ResolveNight();Render();}
  }
  public void Choose(DayAction action)
  {
   if(menu||line<intro.lines.Length)return;
   if(session.TryChoose(action,out var feedback)){message=feedback;Render();}else notice.text=feedback;
  }
  private string Ending(NightOutcome outcome)
  {
   switch(outcome){
    case NightOutcome.SharedSecret:return "【雨夜同谋】\n林月把名册放到灯下：我弟弟也在上面。她留下后山钥匙，决定推迟离山。\n你们约定，明夜子时查清真相。";
    case NightOutcome.LoneWitness:return "【孤证】\n名册的墨迹未干，林月的车却已消失在雨里。\n你有证据，却不知道该相信谁。守阁人恐怕也已记住你的脸。";
    case NightOutcome.GuardedGate:return "【门外剑声】\n你凭新学的剑诀挡住一次试探。黑衣人退进雨幕：登仙之前，别让他们知道你能反抗。\n你活过第一夜，却仍不明白他为何而来。";
    default:return "【不安之夜】\n夜半车轮碾过山道，钟声掩住一声呼喊。你没有看清红榜背后的秘密。\n灵石握在掌心，你决定明日不能再错过。";
   }
  }
  private void Render()
  {
   var s=session.Snapshot();status.text="剩余行动 "+s.actionsLeft+"/2    修为 "+s.cultivation+"    林月信任 "+s.trust+"    灵石 "+s.spiritStones+"    证据："+(s.hasEvidence?"旧名册":"无");
   foreach(var b in actions)b.gameObject.SetActive(false);next.gameObject.SetActive(true);next.GetComponentInChildren<Text>().text="继续";
   if(menu){title.text="距离登仙大典，还有三十日";body.text="你可以成为仙人，也可能成为代价。\n\n在有限的白昼里，选择力量、真相，或一个愿意相信你的人。\n当前原型仅包含第一日短流程。";next.GetComponentInChildren<Text>().text="开始新游戏";return;}
   if(line<intro.lines.Length){title.text="第一日 / 山门";body.text=intro.lines[line];return;}
   if(s.resolved){title.text="第一日 / 夜";body.text=Ending(s.outcome);next.GetComponentInChildren<Text>().text="重玩第一日";return;}
   title.text=s.actionsLeft==2?"白昼 / 你打算去哪里？":"暮色 / 每个选择都有代价";
   body.text=string.IsNullOrEmpty(message)?"每项行动消耗一次机会，同一地点今日只能访问一次。\n林月今夜离山，名册也将在夜里移交。你无法把一切都准备妥当。":message;
   if(s.actionsLeft==0){next.GetComponentInChildren<Text>().text="入夜 / 查看结果";return;}
   next.gameObject.SetActive(false);
   for(int i=0;i<4;i++){actions[i].gameObject.SetActive(true);actions[i].interactable=session.CanChoose((DayAction)i,out _);}
  }
  public void Save()
  {
   if(menu){notice.text="请先开始游戏。";return;}
   try{
    var data=new PrototypeSave{version=1,session=session.ToJson(),line=line,message=message};
    Directory.CreateDirectory(Path.GetDirectoryName(SavePath));File.WriteAllText(SavePath+".tmp",JsonUtility.ToJson(data,true));
    if(File.Exists(SavePath))File.Replace(SavePath+".tmp",SavePath,SavePath+".bak");else File.Move(SavePath+".tmp",SavePath);
    notice.text="已保存当前进度。";
   }catch(Exception ex){notice.text="保存失败："+ex.Message;}
  }
  private void RestoreFile(string path)
  {
   var data=JsonUtility.FromJson<PrototypeSave>(File.ReadAllText(path));
   if(data==null||data.version!=1||data.line<0||data.line>intro.lines.Length)throw new Exception("Invalid save");
   var restored=FirstDaySession.Restore(data.session);
   if(data.line<intro.lines.Length&&restored.Snapshot().actionsLeft!=2)throw new Exception("Inconsistent progress");
   session=restored;line=data.line;message=data.message??string.Empty;menu=false;Render();
  }
  public void Load()
  {
   try{RestoreFile(SavePath);notice.text="已读取存档。";}
   catch(Exception){try{RestoreFile(SavePath+".bak");notice.text="主存档不可读，已恢复备份。";}catch(Exception){notice.text="没有有效存档；当前进度未改变。";}}
  }
  private void OnDestroy(){if(font!=null)Destroy(font);}
 }
}
